cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0123 NEW)
project(whu_chassis LANGUAGES C CXX)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CHASSIS_BUILD "host" CACHE STRING "Build family: host or firmware")
set_property(CACHE CHASSIS_BUILD PROPERTY STRINGS host firmware)
include(cmake/ChassisMpc.cmake)

if(CHASSIS_BUILD STREQUAL "host")
  if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Host validation requires a native compiler; use a separate build directory.")
  endif()
  include(CTest)
  add_subdirectory(tools/chassis/mpc)
  if(BUILD_TESTING)
    add_executable(chassis_follow_checks
      tools/chassis/tests/chassis_follow_checks.c
      src/module/chassis/chassis_follow.c)
    target_include_directories(chassis_follow_checks PRIVATE src/module/chassis)
    target_compile_features(chassis_follow_checks PRIVATE c_std_99)
    if(NOT WIN32)
      target_link_libraries(chassis_follow_checks PRIVATE m)
    endif()
    add_test(NAME chassis_follow_checks COMMAND chassis_follow_checks)
    add_executable(chassis_motion_checks
      tools/chassis/tests/chassis_motion_checks.c
      src/module/chassis/chassis_motion.c
      src/components/controller/chassis_balance.c)
    target_include_directories(chassis_motion_checks PRIVATE src/module/chassis src/components/controller)
    target_compile_features(chassis_motion_checks PRIVATE c_std_99)
    if(NOT WIN32)
      target_link_libraries(chassis_motion_checks PRIVATE m)
    endif()
    add_test(NAME chassis_motion_checks COMMAND chassis_motion_checks)
    add_executable(chassis_leg_math_checks
      tools/chassis/tests/chassis_leg_math_checks.c
      src/components/algorithm/chassis_leg_math.c)
    target_include_directories(chassis_leg_math_checks PRIVATE src/components/algorithm)
    target_compile_features(chassis_leg_math_checks PRIVATE c_std_99)
    if(NOT WIN32)
      target_link_libraries(chassis_leg_math_checks PRIVATE m)
    endif()
    add_test(NAME chassis_leg_math_checks COMMAND chassis_leg_math_checks)
  endif()
elseif(CHASSIS_BUILD STREQUAL "firmware")
  if(NOT CMAKE_C_COMPILER_ID STREQUAL "ARMClang")
    message(FATAL_ERROR "Firmware currently requires cmake/toolchains/armclang.cmake (legacy ARM libraries).")
  endif()
  include(cmake/Firmware.cmake)
else()
  message(FATAL_ERROR "Unknown CHASSIS_BUILD=${CHASSIS_BUILD}")
endif()
